Playbook Skeletonplay
---
- name: Configure web servers
  hosts: webservers
  become: yes
  gather_facts: yes
  vars:
    http_port: 80

  tasks:
    - name: Install packages
      ansible.builtin.apt:
        name: nginx
        state: present

  handlers:
    - name: Restart nginx
      ansible.builtin.service:
        name: nginx
        state: restarted
Variables Sectionvars
  vars:
    http_port: 80
    https_port: 443
    server_name: webserver01
    max_connections: 1000
    enable_ssl: true
    packages:
      - httpd
      - mod_ssl
      - firewalld
    config_paths:
      apache: /etc/httpd/conf/httpd.conf
      ssl: /etc/httpd/conf.d/ssl.conf
External Variable Filesvars
  vars_files:
    - vars/common.yml
    - vars/{{ ansible_os_family }}.yml
    - vars/secrets.yml
Interactive Promptsvars
  vars_prompt:
    - name: username
      prompt: "Enter the username"
      private: no
    - name: password
      prompt: "Enter the password"
      private: yes
      encrypt: sha512_crypt
Basic Tasktask
    - name: Describe this task
      ansible.builtin.module_name:
        param1: value1
        param2: value2
      register: result
      notify: Handler name
Handlerhandler
    - name: Restart service
      ansible.builtin.service:
        name: service_name
        state: restarted
      listen: "restart service"
Loop Taskloop
    - name: Install multiple packages
      ansible.builtin.apt:
        name: "{{ item }}"
        state: present
      loop:
        - nginx
        - python3
        - git
Loop with Dictionaryloop
    - name: Create multiple users
      ansible.builtin.user:
        name: "{{ item.name }}"
        groups: "{{ item.groups }}"
        state: present
      loop:
        - { name: 'alice', groups: 'wheel' }
        - { name: 'bob', groups: 'developers' }
        - { name: 'carol', groups: 'developers,wheel' }
      loop_control:
        label: "{{ item.name }}"
Conditional Taskwhen
    - name: Run only on Ubuntu
      ansible.builtin.command: uname -a
      when: ansible_os_family == "Debian"
      register: output
      changed_when: false
Block / Rescueblock
    - name: Handle errors gracefully
      block:
        - name: Attempt risky task
          ansible.builtin.command: /opt/risky.sh
      rescue:
        - name: Recovery task
          ansible.builtin.debug:
            msg: "Task failed, recovering"
      always:
        - name: Always run cleanup
          ansible.builtin.debug:
            msg: "Cleanup complete"
Register & Debugtask
    - name: Check service status
      ansible.builtin.command: systemctl status httpd
      register: service_result
      changed_when: false
      failed_when: false

    - name: Display result
      ansible.builtin.debug:
        msg: "Status: {{ service_result.rc }}"

    - name: Show full output
      ansible.builtin.debug:
        var: service_result.stdout_lines
File Managementtask
    - name: Create directory structure
      ansible.builtin.file:
        path: /opt/app/config
        state: directory
        owner: appuser
        group: appgroup
        mode: '0755'
        recurse: yes

    - name: Copy configuration file
      ansible.builtin.copy:
        src: files/app.conf
        dest: /opt/app/config/app.conf
        owner: appuser
        group: appgroup
        mode: '0644'
        backup: yes
      notify: Restart application
Template Tasktask
    - name: Deploy configuration file
      ansible.builtin.template:
        src: templates/config.j2
        dest: /etc/app/config.conf
        owner: root
        group: root
        mode: '0644'
      notify: Restart app
Line in Filetask
    - name: Ensure line exists in config
      ansible.builtin.lineinfile:
        path: /etc/ssh/sshd_config
        regexp: '^PermitRootLogin'
        line: 'PermitRootLogin no'
        state: present
        backup: yes
      notify: Restart sshd

    - name: Add line after match
      ansible.builtin.lineinfile:
        path: /etc/hosts
        insertafter: '^127\.0\.0\.1'
        line: '192.168.1.100 appserver'
User Managementtask
    - name: Create application user
      ansible.builtin.user:
        name: appuser
        comment: "Application Service Account"
        uid: 1050
        group: appgroup
        groups: wheel,developers
        append: yes
        shell: /bin/bash
        home: /home/appuser
        create_home: yes
        state: present

    - name: Create application group
      ansible.builtin.group:
        name: appgroup
        gid: 1050
        state: present
Service Managementhandler
    - name: Ensure service is running and enabled
      ansible.builtin.systemd:
        name: httpd
        state: started
        enabled: yes
        daemon_reload: yes

    - name: Wait for service to be ready
      ansible.builtin.wait_for:
        port: 80
        delay: 5
        timeout: 30
Firewall Rulestask
    - name: Open HTTP port in firewall
      ansible.posix.firewalld:
        port: "{{ http_port }}/tcp"
        permanent: yes
        immediate: yes
        state: enabled

    - name: Allow HTTPS service
      ansible.posix.firewalld:
        service: https
        permanent: yes
        immediate: yes
        state: enabled
Set Factsvars
    - name: Set deployment timestamp
      ansible.builtin.set_fact:
        deploy_time: "{{ ansible_date_time.iso8601 }}"
        app_version: "2.1.0"

    - name: Build dynamic config
      ansible.builtin.set_fact:
        app_url: "https://{{ server_name }}:{{ https_port }}"
        log_path: "/var/log/{{ app_name }}/app.log"
Include Variablesvars
    - name: Load OS-specific variables
      ansible.builtin.include_vars:
        file: "vars/{{ ansible_os_family }}.yml"

    - name: Load variables from directory
      ansible.builtin.include_vars:
        dir: vars/overrides
        extensions:
          - yml
          - yaml
Include Rolerole
    - name: Apply database role
      ansible.builtin.include_role:
        name: database
      vars:
        db_name: production
        db_user: admin
Multiple Handlershandler
  handlers:
    - name: Restart httpd
      ansible.builtin.systemd:
        name: httpd
        state: restarted

    - name: Restart firewalld
      ansible.builtin.systemd:
        name: firewalld
        state: restarted

    - name: Reload sshd
      ansible.builtin.systemd:
        name: sshd
        state: reloaded

    - name: Run post-deploy script
      ansible.builtin.command: /opt/app/post-deploy.sh
      listen: "deploy complete"
Assert & Validatetask
    - name: Validate required variables
      ansible.builtin.assert:
        that:
          - http_port is defined
          - http_port | int > 0
          - server_name | length > 0
        fail_msg: "Required variables are missing"
        success_msg: "All variables validated"
Tags Exampletask
    - name: Install packages
      ansible.builtin.dnf:
        name: "{{ item }}"
        state: present
      loop: "{{ packages }}"
      tags:
        - install
        - packages
        - never

    - name: Configure application
      ansible.builtin.template:
        src: app.conf.j2
        dest: /etc/app/app.conf
      tags:
        - configure
        - always
Blockinfiletask
    - name: Add config block to file
      ansible.builtin.blockinfile:
        path: /etc/httpd/conf/httpd.conf
        marker: "# {mark} ANSIBLE MANAGED BLOCK"
        block: |
          ServerName {{ server_name }}
          Listen {{ http_port }}
          MaxClients {{ max_connections }}
        backup: yes
      notify: Restart httpd
Cron Jobtask
    - name: Schedule log rotation
      ansible.builtin.cron:
        name: "Rotate application logs"
        minute: "0"
        hour: "2"
        job: "/usr/sbin/logrotate /etc/logrotate.d/app"
        user: root
        state: present
YAML Indentation Rules
Spaces onlyNever use tabs. Ansible uses 2-space indent.
ConsistentSame indent level for siblings in a list or map.
List items- marker counts as indent. Content aligns after "- ".
Nested mapsEach nesting adds 2 spaces.
Common Structures
key: valueBasic key-value pair
- itemList/sequence item
key:Map with nested content below
key: |Literal block scalar (preserves newlines)
key: >Folded scalar (joins lines)
&anchorDefine anchor for reuse
*anchorReference an anchor
---Document start marker
Ansible Directives
hosts:Target host group
become:Privilege escalation (yes/no)
gather_facts:Collect system info (yes/no)
vars:Define play/task variables
tasks:List of tasks to execute
handlers:Triggered tasks (notify)
roles:Include roles
when:Conditional execution
loop:Iterate over items
register:Store task output
notify:Trigger handler
tags:Label tasks for selective run
